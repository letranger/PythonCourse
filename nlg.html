註：Keras 3 已移除 tensorflow.keras.preprocessing.text.Tokenizer 與 tensorflow.keras.preprocessing.sequence.pad_sequences=。現行對應做法：Tokenizer 的功能改由 =keras.layers.TextVectorization 承接（先 adapt() 建立詞彙表，再把文字轉成整數序列）；=pad_sequences= 則移至 =keras.utils.pad_sequences=。以下範例已依新版 API 改寫。

 1: import keras
 2: from keras.layers import TextVectorization
 3: from keras.utils import pad_sequences, to_categorical
 4: import numpy as np
 5: 
 6: sentence = "AI is the new electrcity \n AI is my power"
 7: corpus = sentence.lower().split("\n") #converting the sentence to lowercase
 8: 
 9: # 用 TextVectorization 建立詞彙表（取代 Keras 3 移除的 Tokenizer）
10: vectorizer = TextVectorization()
11: vectorizer.adapt(corpus) #creates tokens for each words
12: total_words = vectorizer.vocabulary_size() #calculating total number of words in the initial sentence
13: 
14: input_sequences = [] #training features (x) will be a list
15: 
16: for line in corpus:
17:         token_list = [int(t) for t in vectorizer([line]).numpy()[0] if t != 0] #converts each sentence as its tokenized equivalent
18:         for i in range(1, len(token_list)):
19:                 n_gram_sequence = token_list[:i+1] #generating n gram sequences
20:                 input_sequences.append(n_gram_sequence) #appending each n gram sequence to the list of our features (xs)
21: 
22: max_sequence_len = max([len(x) for x in input_sequences]) #calculating the length of the longest sequence
23: input_sequences = np.array(pad_sequences(input_sequences, maxlen=max_sequence_len, padding='pre')) #pre-pading each value of the input_sequence
24: xs, labels = input_sequences[:,:-1],input_sequences[:,-1] #creating xs and their labels using numpy slicing
25: ys = to_categorical(labels, num_classes=total_words) #creating one hot encoding values
26: 
27: print("詞彙量:", total_words)
28: print("訓練特徵 xs 形狀:", xs.shape)
29: print("one-hot 標籤 ys 形狀:", ys.shape)
