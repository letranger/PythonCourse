2. 統計錄影時間

 1: from os import listdir
 2: from os.path import join
 3: # 註：MoviePy 2.x 已移除 moviepy.editor 模組，改為直接從 moviepy 匯入
 4: from moviepy import VideoFileClip
 5: 
 6: def caltime(path):
 7:     files = listdir(path)
 8: 
 9:     tot = 0
10:     for f in files:
11:         if f.endswith("mp4"):
12:            fullpath = join(path, f)
13:            clip = VideoFileClip(fullpath)
14:            tot += clip.duration
15:            print(clip.duration/60,": ", fullpath)  # 秒數換算為分鐘
16:     return tot/3600
17:

19: print("入門: %.2f小時" %(caltime(path)))
